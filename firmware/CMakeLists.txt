cmake_minimum_required(VERSION 3.13)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(qulay_acquisition C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
pico_sdk_init()
add_executable(qulay_acquisition src/main.c)
target_include_directories(qulay_acquisition PRIVATE include)
target_link_libraries(qulay_acquisition pico_stdlib hardware_spi hardware_irq hardware_sync)
target_compile_options(qulay_acquisition PRIVATE -Wall -Wextra)
pico_enable_stdio_usb(qulay_acquisition 1)
pico_enable_stdio_uart(qulay_acquisition 0)
pico_add_extra_outputs(qulay_acquisition)
